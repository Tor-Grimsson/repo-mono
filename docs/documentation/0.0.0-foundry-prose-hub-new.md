### Step 1: Fix Tag Component
**File:** `/packages/ui/src/atoms/Tag.jsx`

Add `onClick` prop and pass it to the div:

```jsx
export default function Tag({
  text,
  children,
  variant = 'default',
  className = '',
  onClick  // ADD THIS
}) {
  const baseClass = variant === 'inverse' ? 'control-unified-inverse' : 'tag-control'
  const content = children || text

  return (
    <div
      className={`cursor-pointer ${baseClass} whitespace-nowrap ${className}`}
      onClick={onClick}  // ADD THIS
    >
      <span>{content}</span>
    </div>
  )
}
```
